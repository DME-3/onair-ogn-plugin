cmake_minimum_required(VERSION 3.20)
project(onair_ogn VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
option(OGN_SANITIZE "Run tests with address/undefined-behaviour sanitizers" OFF)
set(OGN_SANITIZERS "address,undefined" CACHE STRING "Clang sanitizers to enable")
add_compile_options(-Wall -Wextra -Wpedantic "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-fmacro-prefix-map=${CMAKE_SOURCE_DIR}=.")
if(OGN_SANITIZE)
  add_compile_options(-fsanitize=${OGN_SANITIZERS} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${OGN_SANITIZERS})
endif()
add_library(ogn_core STATIC src/ldpc.cpp src/protocol.cpp src/receiver.cpp
  vendor/onair/src/def.cpp vendor/onair/src/bits.cpp vendor/onair/src/frames.cpp
  vendor/onair/src/fec.cpp vendor/onair/src/rx.cpp vendor/onair/src/json_mini.cpp)
target_include_directories(ogn_core PUBLIC src vendor/onair/include)
set_target_properties(ogn_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(ogn-mband SHARED src/plugin.cpp)
target_link_libraries(ogn-mband PRIVATE ogn_core)
target_include_directories(ogn-mband PRIVATE sdk)
set_target_properties(ogn-mband PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/ogn-mband")
if(APPLE)
  target_link_options(ogn-mband PRIVATE "LINKER:-exported_symbol,_onair_plugin_entry")
endif()
add_custom_command(TARGET ogn-mband POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  ${CMAKE_SOURCE_DIR}/manifest.json ${CMAKE_BINARY_DIR}/plugins/ogn-mband/manifest.json)
enable_testing()
add_library(ogn_generator STATIC vendor/onair/src/gen.cpp)
target_include_directories(ogn_generator PUBLIC vendor/onair/include)
target_link_libraries(ogn_generator PUBLIC ogn_core)
add_executable(ogn_tests tests/test_ogn.cpp)
target_link_libraries(ogn_tests PRIVATE ogn_generator)
add_test(NAME ogn_protocol_and_iq COMMAND ogn_tests)
add_executable(ogn_replay tests/replay_host.cpp)
target_include_directories(ogn_replay PRIVATE sdk)
target_link_libraries(ogn_replay PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME ogn_plugin_load COMMAND ogn_replay $<TARGET_FILE:ogn-mband> --self-test)
add_test(NAME ogn_generate_fixture COMMAND ogn_tests --fixture ${CMAKE_BINARY_DIR}/synthetic.cf32)
set_tests_properties(ogn_generate_fixture PROPERTIES FIXTURES_SETUP iq)
add_test(NAME ogn_plugin_iq COMMAND ogn_replay $<TARGET_FILE:ogn-mband> --input ${CMAKE_BINARY_DIR}/synthetic.cf32 --format cf32 --rate 1250000 --centre 868300000 --expect-positions 4 --expect-public-vector)
add_test(NAME ogn_plugin_m1 COMMAND ogn_replay $<TARGET_FILE:ogn-mband> --input ${CMAKE_BINARY_DIR}/synthetic.cf32 --format cf32 --rate 1250000 --centre 868300000 --mode m1 --expect-positions 2 --chunk 777 --expect-public-vector)
add_test(NAME ogn_plugin_m2_override COMMAND ogn_replay $<TARGET_FILE:ogn-mband> --input ${CMAKE_BINARY_DIR}/synthetic.cf32 --format cf32 --rate 1250000 --centre 868400000 --recording-centre-mhz 868.3 --mode m2 --expect-positions 2 --expect-public-vector)
set_tests_properties(ogn_plugin_iq ogn_plugin_m1 ogn_plugin_m2_override PROPERTIES FIXTURES_REQUIRED iq)

add_test(NAME ogn_generate_missing COMMAND ogn_tests --missing-fixture ${CMAKE_BINARY_DIR}/missing.cf32)
set_tests_properties(ogn_generate_missing PROPERTIES FIXTURES_SETUP missing_iq)
add_test(NAME ogn_plugin_missing_fields COMMAND ogn_replay $<TARGET_FILE:ogn-mband> --input ${CMAKE_BINARY_DIR}/missing.cf32 --format cf32 --rate 1250000 --centre 868300000 --mode m1 --expect-positions 1 --expect-unavailable)
set_tests_properties(ogn_plugin_missing_fields PROPERTIES FIXTURES_REQUIRED missing_iq)
